# 数据库建表语句

## 1. 创建数据库

```sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS pharmacy_inventory CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- 使用数据库
USE pharmacy_inventory;
```

## 2. 建表语句

### 2.1 category（药品分类）表

```sql
CREATE TABLE IF NOT EXISTS category (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    name VARCHAR(50) NOT NULL COMMENT '分类名称',
    parent_id BIGINT DEFAULT 0 COMMENT '父分类ID（0表示一级分类）',
    level INT DEFAULT 1 COMMENT '分类级别',
    description VARCHAR(500) COMMENT '分类描述',
    sort INT DEFAULT 0 COMMENT '排序值',
    status INT DEFAULT 1 COMMENT '状态：0-禁用，1-启用',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='药品分类表';
```

### 2.2 symptom（症状）表

```sql
CREATE TABLE IF NOT EXISTS symptom (
    id INT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    name VARCHAR(255) NOT NULL COMMENT '症状名称',
    description TEXT COMMENT '症状描述',
    PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='症状表';
```

### 2.3 medicine（药品信息）表

```sql
CREATE TABLE IF NOT EXISTS medicine (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    medicine_code VARCHAR(50) NOT NULL UNIQUE COMMENT '药品编码（唯一）',
    name VARCHAR(100) NOT NULL COMMENT '药品名称',
    generic_name VARCHAR(100) COMMENT '通用名称',
    category_id BIGINT COMMENT '分类ID',
    specification VARCHAR(200) COMMENT '药品规格',
    unit VARCHAR(20) COMMENT '单位（盒、瓶、支等）',
    manufacturer VARCHAR(200) COMMENT '生产厂家',
    approval_number VARCHAR(100) COMMENT '批准文号',
    description TEXT COMMENT '药品描述',
    retail_price DECIMAL(10,2) COMMENT '零售价',
    purchase_price DECIMAL(10,2) COMMENT '采购价',
    status INT DEFAULT 1 COMMENT '状态：0-停用，1-启用',
    is_seasonal BOOLEAN DEFAULT FALSE COMMENT '是否为季节性药品',
    is_prescription BOOLEAN DEFAULT FALSE COMMENT '是否为处方药',
    storage_requirement INT COMMENT '存储要求',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (id),
    FOREIGN KEY (category_id) REFERENCES category(id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='药品信息表';
```

### 2.4 user（用户信息）表

```sql
CREATE TABLE IF NOT EXISTS user (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名（唯一）',
    password VARCHAR(255) NOT NULL COMMENT '密码（加密存储）',
    real_name VARCHAR(50) COMMENT '真实姓名',
    phone VARCHAR(20) COMMENT '手机号',
    email VARCHAR(100) COMMENT '邮箱',
    role VARCHAR(20) NOT NULL COMMENT '角色：ADMIN, PHARMACIST, PURCHASER',
    status INT DEFAULT 1 COMMENT '状态：0-禁用，1-正常',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='用户信息表';
```

### 2.5 stock（库存信息）表

```sql
CREATE TABLE IF NOT EXISTS stock (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    medicine_id BIGINT NOT NULL COMMENT '药品ID',
    batch_number VARCHAR(50) COMMENT '批号',
    production_date DATE COMMENT '生产日期',
    expiration_date DATE COMMENT '有效期至',
    quantity INT NOT NULL DEFAULT 0 COMMENT '当前数量',
    warning_quantity INT DEFAULT 10 COMMENT '库存预警数量',
    shelf_location VARCHAR(50) COMMENT '货架位置',
    status INT DEFAULT 1 COMMENT '状态：0-过期，1-正常',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    minimum_order_quantity INT COMMENT '最小订购数量',
    lead_time_days INT COMMENT '采购提前期（天）',
    reorder_point INT COMMENT '再订货点',
    PRIMARY KEY (id),
    FOREIGN KEY (medicine_id) REFERENCES medicine(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='库存信息表';
```

### 2.6 purchase_order（采购订单）表

```sql
CREATE TABLE IF NOT EXISTS purchase_order (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    order_no VARCHAR(50) NOT NULL UNIQUE COMMENT '订单编号（唯一）',
    medicine_id BIGINT NOT NULL COMMENT '药品ID',
    quantity INT NOT NULL COMMENT '采购数量',
    unit_price DECIMAL(10,2) NOT NULL COMMENT '采购单价',
    total_amount DECIMAL(10,2) NOT NULL COMMENT '总金额',
    supplier VARCHAR(200) COMMENT '供应商',
    order_status INT NOT NULL DEFAULT 0 COMMENT '订单状态：0-待处理，1-已确认，2-已到货，3-已取消',
    order_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '下单时间',
    expected_arrival DATE COMMENT '预计到货日期',
    actual_arrival DATETIME COMMENT '实际到货时间',
    operator_id BIGINT COMMENT '操作员ID',
    remark VARCHAR(500) COMMENT '备注',
    PRIMARY KEY (id),
    FOREIGN KEY (medicine_id) REFERENCES medicine(id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (operator_id) REFERENCES user(id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='采购订单表';
```

### 2.7 sale_record（销售记录）表

```sql
CREATE TABLE IF NOT EXISTS sale_record (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    record_no VARCHAR(50) NOT NULL UNIQUE COMMENT '销售单号（唯一）',
    medicine_id BIGINT NOT NULL COMMENT '药品ID',
    quantity INT NOT NULL COMMENT '销售数量',
    unit_price DECIMAL(10,2) NOT NULL COMMENT '销售单价',
    total_amount DECIMAL(10,2) NOT NULL COMMENT '总金额',
    customer_info VARCHAR(200) COMMENT '顾客信息',
    customer_type INT COMMENT '顾客类型',
    is_Rx BOOLEAN DEFAULT FALSE COMMENT '是否为处方药销售',
    sale_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '销售时间',
    operator_id BIGINT COMMENT '操作员ID',
    remark VARCHAR(500) COMMENT '备注',
    PRIMARY KEY (id),
    FOREIGN KEY (medicine_id) REFERENCES medicine(id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (operator_id) REFERENCES user(id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='销售记录表';
```

### 2.8 prediction_result（预测结果）表

```sql
CREATE TABLE IF NOT EXISTS prediction_result (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    medicine_id BIGINT NOT NULL COMMENT '药品ID',
    prediction_date DATE NOT NULL COMMENT '预测日期',
    predicted_quantity INT NOT NULL COMMENT '预测需求量',
    confidence_interval_lower INT COMMENT '置信区间下限',
    confidence_interval_upper INT COMMENT '置信区间上限',
    model_type VARCHAR(50) COMMENT '预测模型类型（Prophet, ARIMA, LightGBM等）',
    accuracy_rate DECIMAL(5,2) COMMENT '预测准确率（百分比）',
    recommended_order_quantity INT COMMENT '建议订购数量',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (id),
    FOREIGN KEY (medicine_id) REFERENCES medicine(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='预测结果表';
```

### 2.9 medicine_symptom（药品症状关联）表

```sql
CREATE TABLE IF NOT EXISTS medicine_symptom (
    medicine_id BIGINT NOT NULL COMMENT '药品ID',
    symptom_id INT NOT NULL COMMENT '症状ID',
    PRIMARY KEY (medicine_id, symptom_id),
    FOREIGN KEY (medicine_id) REFERENCES medicine(id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (symptom_id) REFERENCES symptom(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='药品症状关联表';
```

### 2.10 sale_record_symptom（销售记录症状关联）表

```sql
CREATE TABLE IF NOT EXISTS sale_record_symptom (
    sale_record_id BIGINT NOT NULL COMMENT '销售记录ID',
    symptom_id INT NOT NULL COMMENT '症状ID',
    PRIMARY KEY (sale_record_id, symptom_id),
    FOREIGN KEY (sale_record_id) REFERENCES sale_record(id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (symptom_id) REFERENCES symptom(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='销售记录症状关联表';
```

## 3. 索引优化

```sql
-- 为常用查询字段添加索引

-- category表索引
CREATE INDEX idx_category_parent_id ON category(parent_id);
CREATE INDEX idx_category_level ON category(level);
CREATE INDEX idx_category_status ON category(status);

-- medicine表索引
CREATE INDEX idx_medicine_category_id ON medicine(category_id);
CREATE INDEX idx_medicine_status ON medicine(status);
CREATE INDEX idx_medicine_medicine_code ON medicine(medicine_code);

-- stock表索引
CREATE INDEX idx_stock_medicine_id ON stock(medicine_id);
CREATE INDEX idx_stock_batch_number ON stock(batch_number);
CREATE INDEX idx_stock_expiration_date ON stock(expiration_date);
CREATE INDEX idx_stock_status ON stock(status);

-- purchase_order表索引
CREATE INDEX idx_purchase_order_medicine_id ON purchase_order(medicine_id);
CREATE INDEX idx_purchase_order_order_status ON purchase_order(order_status);
CREATE INDEX idx_purchase_order_operator_id ON purchase_order(operator_id);
CREATE INDEX idx_purchase_order_order_time ON purchase_order(order_time);

-- sale_record表索引
CREATE INDEX idx_sale_record_medicine_id ON sale_record(medicine_id);
CREATE INDEX idx_sale_record_operator_id ON sale_record(operator_id);
CREATE INDEX idx_sale_record_sale_time ON sale_record(sale_time);

-- prediction_result表索引
CREATE INDEX idx_prediction_result_medicine_id ON prediction_result(medicine_id);
CREATE INDEX idx_prediction_result_prediction_date ON prediction_result(prediction_date);

-- user表索引
CREATE INDEX idx_user_username ON user(username);
CREATE INDEX idx_user_role ON user(role);
CREATE INDEX idx_user_status ON user(status);
```

## 5. 使用说明

1. **执行顺序**：
   - 首先执行创建数据库和使用数据库的语句
   - 然后按照表的依赖关系执行建表语句（先创建被依赖的表）
   - 接着执行索引优化语句
   - 最后执行初始数据插入语句

2. **注意事项**：
   - 本脚本使用utf8mb4字符集，确保数据库支持该字符集
   - 所有表均使用InnoDB引擎，支持事务和外键
   - 密码字段使用BCrypt加密存储，示例密码已加密
   - 初始数据仅作为示例，可根据实际需求修改

3. **数据库连接配置**：
   - 数据库名称：pharmacy_inventory
   - 用户名：root
   - 密码：114514aA@（根据实际配置修改）
   - 端口：3306（默认）

4. **后续维护**：
   - 定期备份数据库
   - 根据业务增长情况调整表结构和索引
   - 监控数据库性能，必要时进行优化

## 6. 故障排查

### 6.1 外键约束错误

如果执行建表语句时遇到外键约束错误，可能是因为表的创建顺序不正确。请按照以下顺序创建表：

1. category
2. symptom
3. medicine
4. user
5. stock
6. purchase_order
7. sale_record
8. prediction_result
9. medicine_symptom
10. sale_record_symptom

### 6.2 字符集错误

如果遇到字符集错误，请确保数据库服务器支持utf8mb4字符集。可以通过以下命令检查：

```sql
SHOW VARIABLES LIKE 'character_set_%';
SHOW VARIABLES LIKE 'collation_%';
```

### 6.3 索引创建错误

如果遇到索引创建错误，可能是因为索引名称重复或长度超过限制。请检查索引名称是否唯一，并确保长度在限制范围内。
